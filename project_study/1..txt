<template>
  <div>
    <form>
      邮箱号：<input type="text" :disabled="!isCode" id="email" v-model="email">
      <br>
      验证码：<input type="text" :disabled="isCode" v-model="code">
      <br>
      <button type="button" @click="sendEmail" v-show="isCode">发送验证码</button>
      <button type="button" @click="checkCode" v-show="!isCode">验证验证码</button>
    </form>
  </div>
</template>

<script setup>
import {ref, getCurrentInstance} from "vue";

//控制输入框是否可以输入的变量
let isCode = ref(true);


// 接受邮箱号和验证码的变量
let email = ref("");
let code = ref("");

// 获取当前实例对象 --- 需要通过这个对象才可以访问到main.js中定义的$axios变量
let proxy = getCurrentInstance().proxy;
// 定义发送验证码函数
function sendEmail(){
  let sendEmail = email.value; //获取到用户输入的邮箱账号
  // 通过封装的axios请求访问服务器定义的发送验证码接口
  proxy.$axios({ //{}中写的就是访问服务器接口的信息：比如请求地址【自动拼接前缀hhttp://localhost:8000/】,请求方式【默认get】，请求参数
    url: 'users/sendEmail', //请求地址
    method: 'get', //请求方式
    params: { // 请求参数
      email: sendEmail, //参数的key必须和服务器接口的形参一致
    },
  }).then(res => { // 请求成功的回调函数 --- 即请求服务器接口成功后执行这个代码
      // 这个代码本质上就是一个函数，只使用了函数的简化写法【箭头函数】
      //res 就是函数的形参, => 固定写法， {}内的内容就是函数体
      // res => {}等价于function a(res){}
    console.log(res);
    let code = res.data.code;
    let msg = res.data.msg;
    //逻辑判断处理处理 --- ===表示全等【数据类型和值都相等为true】 == 表示只要值相等就为true
    if (code === 200){
      // 修改isCode变量的值
      isCode.value = !isCode.value;
      alert(msg);
    } else {
      alert(msg);
    }
  });
}

function checkCode(){

  let checkEmail = { // 把输入的邮箱号和验证码封装成一个js对象
    email: email.value,
    code: code.value
  }
  // JSON.stringify(checkCode) -- 把js对象转换成json字符串
  // JSON.parse(json) -- 把json转为js对象
  // axios中使用post请求，设置参数是用data属性，比如data: JSON.stringify(checkCode)
  JSON.stringify(checkCode)

  proxy.$axios({
    url: 'users/Verification_Module',
    method: 'post',
    params:{
      email: email,
      code: code,
    }
  })
}

</script>
